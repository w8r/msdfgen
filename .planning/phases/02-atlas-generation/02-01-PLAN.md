```typescript
/**
 * Configuration for atlas generation
 */
export interface AtlasConfig {
  /** Glyph bitmap size in pixels (default: 32) */
  glyphSize?: number;
  /** Padding between glyphs in pixels (default: 2) */
  padding?: number;
  /** SDF distance range in pixels (default: 4) */
  distanceRange?: number;
}

/**
 * Information about a single glyph in the atlas
 */
export interface GlyphInfo {
  /** Atlas position in pixels */
  atlasBounds: {
    left: number;
    bottom: number;
    right: number;
    top: number;
  };
  /** Normalized UV coordinates (0-1 range, GPU-ready) */
  uvBounds: {
    u0: number;
    v0: number;
    u1: number;
    v1: number;
  };
  /** Glyph bounds in em units (normalized to unitsPerEm) */
  planeBounds: {
    left: number;
    bottom: number;
    right: number;
    top: number;
  };
  /** Advance width in font units */
  advanceWidth: number;
  /** Left side bearing in font units */
  leftSideBearing: number;
}

/**
 * Result of atlas generation
 */
export interface AtlasResult {
  /** The generated MSDF atlas bitmap */
  atlas: Bitmap<Float32Array, 3>;
  /** Glyph information keyed by character */
  glyphs: Map<string, GlyphInfo>;
  /** Time taken to generate the atlas in milliseconds */
  generationTimeMs: number;
  /** Atlas width in pixels */
  atlasWidth: number;
  /** Atlas height in pixels */
  atlasHeight: number;
}
```

- Include both font-unit values (advanceWidth, leftSideBearing) AND atlas-space values (atlasBounds, uvBounds)
- Map keyed by character string

Create `src/atlas/AtlasGenerator.ts` implementing the atlas generation pipeline:

1. **Function signature:**
```typescript
export function generateAtlas(
  font: Font,
  chars: string,
  config: AtlasConfig = {}
): AtlasResult
```

2. **Implementation steps:**

a) **Deduplicate and validate characters:**
   - `[...new Set(chars)]` to get unique characters
   - Filter to only characters that exist in font via `font.hasGlyph(char)`

b) **Get glyph shapes and compute sizes:**
   - For each character, get glyph via `font.getGlyph(char)`
   - Get shape via `glyph.toShape()`
   - Check if shape has contours (skip empty shapes like space from MSDF but keep metrics)
   - Compute required cell size: `glyphSize + padding * 2`

c) **Run potpack:**
   - Create boxes array with `{ w, h, char }` for potpack
   - Call `potpack(boxes)` - it mutates boxes adding `x, y` positions
   - Get result `{ w, h }` for atlas dimensions

d) **Round to power of two:**
   - Helper: `nextPowerOfTwo(n) = 1 << Math.ceil(Math.log2(n))`
   - Apply to both width and height

e) **Allocate atlas bitmap:**
   - `new Bitmap(Float32Array, 3, atlasWidth, atlasHeight)`
   - Fill with default value (0.5 for neutral SDF)

f) **Generate MSDFs and compose:**
   - For each box (with position from potpack):
     - Get the glyph and shape
     - If shape has contours:
       - Create transformation using `SDFTransformation` with `Projection` and `Range`
       - Compute scale to fit glyph in `glyphSize` with `distanceRange` margin
       - Generate MSDF into temporary bitmap
       - Copy to atlas at `(box.x + padding, box.y + padding)`
     - Create GlyphInfo with:
       - atlasBounds: pixel coordinates in atlas
       - uvBounds: normalized 0-1 coordinates (flip Y for OpenGL convention)
       - planeBounds: glyph bounds in em units (divide by unitsPerEm)
       - advanceWidth, leftSideBearing from glyph metrics

g) **Return AtlasResult:**
   - atlas, glyphs Map, generationTimeMs, atlasWidth, atlasHeight

**Key implementation details:**

For SDFTransformation setup:
```typescript
const bounds = shape.getBounds();
const margin = distanceRange;
const availableSize = glyphSize - 2 * margin;
const shapeWidth = bounds.r - bounds.l;
const shapeHeight = bounds.t - bounds.b;
const scale = Math.min(availableSize / shapeWidth, availableSize / shapeHeight);

// Center shape in glyph bitmap
const translateX = margin / scale - bounds.l + (availableSize / scale - shapeWidth) / 2;
const translateY = margin / scale - bounds.b + (availableSize / scale - shapeHeight) / 2;

const projection = new Projection(
  new Vector2(scale, scale),
  new Vector2(translateX, translateY)
);
const range = new Range(-distanceRange / scale, distanceRange / scale);
const transformation = new SDFTransformation(projection, range);
```

For UV coordinates (bottom-left OpenGL origin):
```typescript
uvBounds: {
  u0: contentX / atlasWidth,
  v0: 1 - (contentY + glyphSize) / atlasHeight,  // Flip Y
  u1: (contentX + glyphSize) / atlasWidth,
  v1: 1 - contentY / atlasHeight
}
```

Handle empty glyphs (space):
```typescript
if (shape.contours.length === 0) {
  // Skip MSDF generation, still add to glyphs map with metrics
  // UV bounds can be (0,0,0,0) or point to a blank region
}
```
